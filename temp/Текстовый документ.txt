struct Node
{
    int step{};

    int port{};
    bool final{};

    stdvectorNode outerNodes{};

    stdvectorNode nodes{};

};

void output(stdvectorNode& vec)
{
    for (Node& node vec)
    {
        stdcout  node.step     node.port     node.final  'n';
        if (!node.nodes.empty())
        {
            output(node.nodes);
        }
    }
}

stdvectorNode& searchVector(stdvectorNode& vec, int outPort)
{
    for (auto& node  vec)
    {
        if (node.port == outPort)
        {
            return node.nodes;
        }
        if (!node.nodes.empty())
        {

            searchVector(node.nodes, outPort);
        }
    }

    return vec;
}

stdvectorNode& add(stdvectorNode& vec, int step, int outPort, int port, bool final)
{
    if (vec.empty())
    {
        vec.push_back({step, port, final, &vec, {}});
        return vec;
    }


    searchVector(vec, outPort).push_back({step, port, final, &vec, {}});

    return searchVector(vec, outPort);


}
